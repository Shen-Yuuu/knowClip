# 知微 KnowClip - 数据结构设计

## 一、数据库表设计

### 1. knowledge 表（知识条目主表）

```sql
CREATE TABLE knowledge (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  
  -- 内容
  title TEXT,                          -- 标题（可选，用户自定义或自动生成）
  content TEXT,                        -- OCR 识别文本或用户输入文本
  note TEXT,                           -- 用户笔记
  keywords TEXT,                       -- 关键词（逗号分隔）
  
  -- 来源信息
  source_type TEXT NOT NULL,           -- 来源类型：'share' | 'picker' | 'manual'
  source_app TEXT,                     -- 来源应用（如果可获取）
  source_text TEXT,                    -- 分享时的原始文本
  source_url TEXT,                     -- 链接
  
  -- 处理状态
  processing_status TEXT NOT NULL DEFAULT 'pending',  -- 'pending' | 'processing' | 'completed' | 'failed'
  ocr_confidence REAL,                 -- OCR 置信度 (0-1)
  error_message TEXT,                  -- 错误信息（失败时）
  retry_count INTEGER DEFAULT 0,       -- 重试次数
  
  -- 元数据
  is_starred INTEGER DEFAULT 0,        -- 是否标星 (0 | 1)
  is_archived INTEGER DEFAULT 0,       -- 是否归档 (0 | 1)
  created_at INTEGER NOT NULL,         -- 创建时间（毫秒时间戳）
  updated_at INTEGER NOT NULL,         -- 修改时间（毫秒时间戳）
  ocr_completed_at INTEGER,            -- OCR 完成时间
  
  -- 附件信息
  has_image INTEGER DEFAULT 0,         -- 是否有图片附件 (0 | 1)
  image_count INTEGER DEFAULT 0,       -- 图片数量
  
  -- 索引优化
  UNIQUE(content, created_at)          -- 防止完全相同内容在短时间内重复
);

-- 索引
CREATE INDEX idx_created_at ON knowledge(created_at DESC);
CREATE INDEX idx_processing_status ON knowledge(processing_status);
CREATE INDEX idx_starred ON knowledge(is_starred);
CREATE INDEX idx_archived ON knowledge(is_archived);
CREATE INDEX idx_source_type ON knowledge(source_type);
```

### 2. attachments 表（附件表）

```sql
CREATE TABLE attachments (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  knowledge_id INTEGER NOT NULL,       -- 关联的知识条目 ID
  
  -- 文件信息
  file_type TEXT NOT NULL,             -- 文件类型：'image' | 'video' | 'audio' | 'file'
  mime_type TEXT,                      -- MIME 类型
  file_path TEXT NOT NULL,             -- 私有目录中的相对路径
  thumbnail_path TEXT,                 -- 缩略图路径（图片专用）
  
  -- 原始信息
  original_uri TEXT,                   -- 原始 URI（分享时的来源）
  original_name TEXT,                  -- 原始文件名
  
  -- 元数据
  file_size INTEGER,                   -- 文件大小（字节）
  width INTEGER,                       -- 图片宽度
  height INTEGER,                      -- 图片高度
  duration INTEGER,                    -- 视频/音频时长（秒）
  
  -- EXIF 信息（图片专用）
  exif_data TEXT,                      -- EXIF 数据（JSON 字符串）
  taken_at INTEGER,                    -- 拍摄时间（从 EXIF 提取）
  latitude REAL,                       -- 纬度
  longitude REAL,                      -- 经度
  
  -- 时间戳
  created_at INTEGER NOT NULL,
  
  FOREIGN KEY(knowledge_id) REFERENCES knowledge(id) ON DELETE CASCADE
);

-- 索引
CREATE INDEX idx_knowledge_id ON attachments(knowledge_id);
CREATE INDEX idx_file_type ON attachments(file_type);
```

### 4. knowledge_folders 表（知识与文件夹关系表）

```sql
CREATE TABLE knowledge_folders (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  knowledge_id INTEGER NOT NULL,
  folder_id INTEGER NOT NULL,
  created_at INTEGER NOT NULL,
  
  FOREIGN KEY(knowledge_id) REFERENCES knowledge(id) ON DELETE CASCADE,
  FOREIGN KEY(folder_id) REFERENCES folders(id) ON DELETE CASCADE,
  
  UNIQUE(knowledge_id, folder_id)      -- 防止重复关联

-- 索引
CREATE INDEX idx_kf_knowledge_id ON knowledge_folders(knowledge_id);
```

### 5. keywords 表（关键词索引表，用于搜索优化）

```sql
CREATE TABLE keywords (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  knowledge_id INTEGER NOT NULL,
  keyword TEXT NOT NULL,               -- 单个关键词
  weight INTEGER DEFAULT 1,            -- 权重（用于排序）
  
  FOREIGN KEY(knowledge_id) REFERENCES knowledge(id) ON DELETE CASCADE

-- 索引
CREATE INDEX idx_keyword ON keywords(keyword);
CREATE INDEX idx_kw_knowledge_id ON keywords(knowledge_id);
```

### 6. search_history 表（搜索历史）

```sql
CREATE TABLE search_history (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  query TEXT NOT NULL,                 -- 搜索关键词
  result_count INTEGER DEFAULT 0,      -- 结果数量
  searched_at INTEGER NOT NULL,        -- 搜索时间
  
  UNIQUE(query)                        -- 同一关键词只保留最新记录
);

-- 索引
CREATE INDEX idx_searched_at ON search_history(searched_at DESC);
```

## 四、类型守卫与校验

### 1. 数据库结果校验

```typescript
// ❌ 错误：直接使用未校验的查询结果
function getKnowledge(id: number): KnowledgeItem {
  const result = db.querySql(`SELECT * FROM knowledge WHERE id = ?`, [id]);
  return result[0]; // 类型不安全！
}

// ✅ 正确：使用类型守卫
function isValidKnowledgeRow(row: unknown): row is KnowledgeItem {
  if (typeof row !== 'object' || row === null) return false;
  
  const r = row as Record<string, unknown>;
  
  return (
    typeof r.id === 'number' &&
    (typeof r.title === 'string' || r.title === null) &&
    (typeof r.content === 'string' || r.content === null) &&
    typeof r.source_type === 'string' &&
    typeof r.processing_status === 'string' &&
    typeof r.created_at === 'number'
    // ... 更多字段校验
  );
}

function getKnowledge(id: number): KnowledgeItem | null {
  const result = db.querySql(`SELECT * FROM knowledge WHERE id = ?`, [id]);
  
  if (result.length === 0) return null;
  
  const row = result[0];
  if (!isValidKnowledgeRow(row)) {
    hilog.error(DOMAIN, TAG, 'Invalid knowledge row structure');
    return null;
  }
  
  return mapRowToKnowledge(row);
}
```

### 2. JSON 解析校验

```typescript
// ❌ 错误：直接解析
const exifData = JSON.parse(row.exif_data);

// ✅ 正确：安全解析
function parseExifData(jsonString: string | null): Record<string, unknown> | null {
  if (!jsonString) return null;
  
  try {
    const parsed = JSON.parse(jsonString);
    
    if (typeof parsed !== 'object' || parsed === null) {
      return null;
    }
    
    return parsed as Record<string, unknown>;
  } catch (error) {
    hilog.error(DOMAIN, TAG, `Failed to parse EXIF data: ${error}`);
    return null;
  }
}
```

## 五、数据迁移策略

### 版本 1 → 版本 2（示例）

```typescript
const MIGRATIONS: Array<{ version: number; sql: string }> = [
  {
    version: 1,
    sql: `
      -- 初始建表语句
      CREATE TABLE knowledge (...);
      CREATE TABLE attachments (...);
    `
  },
  {
    version: 2,
    sql: `
      -- 添加新字段
      ALTER TABLE knowledge ADD COLUMN ocr_confidence REAL;
      ALTER TABLE knowledge ADD COLUMN retry_count INTEGER DEFAULT 0;
    `
  }
];

function migrateDatabase(db: relationalStore.RdbStore, currentVersion: number): void {
  for (const migration of MIGRATIONS) {
    if (migration.version > currentVersion) {
      hilog.info(DOMAIN, TAG, `Migrating to version ${migration.version}`);
      db.executeSql(migration.sql);
    }
  }
}
```

### 2. 关键词索引优化

```typescript
// 插入知识条目时同步更新关键词索引
function insertKnowledge(item: KnowledgeItem): number {
  db.beginTransaction();
  
  try {
    // 插入主记录
    const knowledgeId = insertKnowledgeRow(item);
    
    // 插入关键词索引
    for (const keyword of item.keywords) {
      db.executeSql(
        'INSERT INTO keywords (knowledge_id, keyword) VALUES (?, ?)',
        [knowledgeId, keyword]
      );
    }
    
    db.commit();
    return knowledgeId;
  } catch (error) {
    db.rollback();
    throw error;
  }
}
```

### 3. 缓存策略

```typescript
// 内存缓存热门搜索结果
const searchCache = new Map<string, { result: SearchResult; timestamp: number }>();
const CACHE_TTL = 5 * 60 * 1000; // 5 分钟

function searchWithCache(query: SearchQuery): SearchResult {
  const cacheKey = JSON.stringify(query);
  const cached = searchCache.get(cacheKey);
  
  if (cached && Date.now() - cached.timestamp < CACHE_TTL) {
    return cached.result;
  }
  
  const result = performSearch(query);
  searchCache.set(cacheKey, { result, timestamp: Date.now() });
  
  return result;
}
```

## 八、数据一致性保障

### 1. 外键约束

所有关联表使用 `FOREIGN KEY` + `ON DELETE CASCADE`，确保删除知识条目时自动清理关联数据。

### 2. 事务保护

所有涉及多表操作的逻辑使用事务包裹。

### 3. 孤立文件清理

定期扫描 `attachments/` 目录，删除数据库中不存在的文件。

```typescript
function cleanOrphanedFiles(): void {
  // 获取数据库中所有附件路径
  const dbPaths = new Set(getAllAttachmentPaths());
  
  // 扫描文件系统
  const fsPaths = scanAttachmentDirectory();
  
  // 删除孤立文件
  for (const fsPath of fsPaths) {
    if (!dbPaths.has(fsPath)) {
      fileIo.unlink(fsPath);
      hilog.info(DOMAIN, TAG, `Deleted orphaned file: ${fsPath}`);
    }
  }
}
```
